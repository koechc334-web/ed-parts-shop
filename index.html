=== AUTH ROUTES === @app.route('/api/register', methods=['POST']) def register(): data = request.json name = data.get('name', '').strip() email = data.get('email', '').lower().strip() password = data.get('password', '') if not name or not email or not password: return jsonify({"error": "Name, email, password required"}), 400 if email in users_db: return jsonify({"error": "Email already exists"}), 400 users_db[email] = { "name": name, "password": generate_password_hash(password), "created": datetime.datetime.now().isoformat() } return jsonify({"message": "Account created", "name": name, "email": email}) @app.route('/api/login', methods=['POST']) def login(): data = request.json email = data.get('email', '').lower().strip() password = data.get('password', '') user = users_db.get(email) if not user or not check_password_hash(user['password'], password): return jsonify({"error": "Wrong email or password"}), 401 return jsonify({"message": "Login ok", "name": user['name'], "email": email}) @app.route('/api/forgot-password', methods=['POST']) def forgot(): email = request.json.get('email', '').lower().strip() if email not in users_db: return jsonify({"message": "If email exists, reset link sent"}), 200 token = secrets.token_urlsafe(32) reset_tokens[token] = {"email": email, "exp": time.time() + 3600} reset_link = f"https://koechc334-web.github.io/ed-parts-shop/reset.html?token={token}" print(f"RESET LINK for {email}: {reset_link}") #
